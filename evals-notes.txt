import json
import ast
import re
from typing import Any, List, Set

def _coerce_to_list(x: Any) -> List[str]:
    """Turn many possible representations into a list of strings."""
    if x is None:
        return []
    if isinstance(x, list):
        return [str(v) for v in x]
    if isinstance(x, dict):
        if "results" in x and isinstance(x["results"], list):
            return [str(v) for v in x["results"]]
        return []
    if isinstance(x, str):
        s = x.strip()
        if not s:
            return []

        # Try direct JSON first
        try:
            obj = json.loads(s)
            return _coerce_to_list(obj)
        except Exception:
            pass

        # Try extracting an embedded JSON object or array
        for pattern in (r"\{.*\}", r"\[.*\]"):
            m = re.search(pattern, s, flags=re.DOTALL)
            if m:
                chunk = m.group(0)
                try:
                    obj = json.loads(chunk)
                    return _coerce_to_list(obj)
                except Exception:
                    try:
                        obj = ast.literal_eval(chunk)
                        return _coerce_to_list(obj)
                    except Exception:
                        pass

        # Last resort: treat as a single label
        return [s]

    return [str(x)]

def _normalize_set(labels: List[str]) -> Set[str]:
    s = {str(x).strip().strip('"').strip("'").lower() for x in labels if str(x).strip()}

    # "Not Applicable" logic
    if "not applicable" in s:
        if len(s) == 1:
            return set()          # NA-only => empty set
        s.discard("not applicable") # NA + others => drop NA

    return s

def _jaccard(a: Set[str], b: Set[str]) -> float:
    union = a | b
    if not union:
        return 1.0  # both empty (e.g., both Not Applicable)
    return len(a & b) / len(union)

def grade(sample, item) -> float:
    """
    Jaccard similarity between:
      - prediction from sample['output_text']
      - ground truth from item['driver']
    """
    pred_raw = sample.get("output_text", "")
    gt_raw = item.get("driver", "")

    pred_list = _coerce_to_list(pred_raw)
    gt_list = _coerce_to_list(gt_raw)

    pred_set = _normalize_set(pred_list)
    gt_set = _normalize_set(gt_list)

    return float(_jaccard(pred_set, gt_set))
